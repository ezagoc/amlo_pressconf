\begin{frame}
\frametitle{Literature Review}

\small
\begin{itemize}
\item Even in the absence of outright censorship, the government still has techniques to silence critics \citep{Censor-Pratt-AER, Autocracy-Guriev-JPE}

\item Some ways in which government can censor in democracies: 

\begin{enumerate}
    \item Bribery: Montesinos in Peru \citep{Peru-Macmillan-JEP}
    \item De-legitimization: South Korea \textit{giraegi} and US (Trump) \citep{KoreaJour-Shin-Journalism, PressCriticism-Carlson-Journalism}.
    \item Restriction of source access: Kisha clubs in Japan \citep{Kisha-Freedom}.
    \item Government advertising in Argentina correlated with favorable coverage \citep{ArgCorr-DiTella-AEJAE}
\end{enumerate}

\item To the best of my knowledge, no causal evidence on any of these.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Research Design RQ1}
\label{slide:lottery}
\small
\begin{itemize}
    \item Estimate the impact of critical coverage on government advertising spending by media outlet. (\hyperlink{slide:intro}{Intro})
    \item OLS probably biased due to time-based endogeneity and unobserved confounding factors related to criticism level and advertising allocation.
    \item \textbf{Instrumental Variable (IV) Approach:} Use lottery assignment to the first, second or third row at morning press conferences as an instrument for $NumCr_{i, t}$.
\item First Stage Equation:

$$NumCr_{i, t} = \theta_0 + \theta_1 FirstRow_{i, t} + \mathbf{X'}_{i,t}\gamma + \epsilon_{i, t}$$
\end{itemize}

\end{frame}
